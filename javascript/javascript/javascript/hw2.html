let dob = prompt("Enter your DOB (YYYY-MM-DD):"); let category = prompt("Enter category:").toLowerCase(); let birthDate = new Date(dob); let today = new Date(); let age = today.getFullYear() - birthDate.getFullYear(); if ( today.getMonth() < birthDate.getMonth() || (today.getMonth() === birthDate.getMonth() && today.getDate() < birthDate.getDate()) ) { age--; } console.log("Your age is: " + age); if (age >= 21) { if (category === "general" || category === "ews") { if (age <= 32) { console.log("Eligible for UPSC"); console.log("Maximum Attempts: 6"); } } if (category === "obc") { if (age <= 35) { console.log("Eligible for UPSC"); console.log("Maximum Attempts: 9"); } } if (category === "sc" || category === "st") { if (age <= 37) { console.log("Eligible for UPSC"); console.log("Unlimited Attempts"); } } if (category === "pwd") { if (age <= 42) { console.log("Eligible for UPSC"); console.log("Age Limit: 42"); } } } if (age < 21) { console.log("Not Eligible for UPSC"); }
