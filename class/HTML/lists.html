import math

def is_prime(n):
    # 1. Handle numbers less than 2
    if n < 2:
        return False
    
    # 2. 2 is the only even prime number
    if n == 2:
        return True
    
    # 3. Eliminate all other even numbers immediately
    if n % 2 == 0:
        return False
    
    # 4. Check odd divisors from 3 up to sqrt(n)
    # We use step=2 to skip even numbers
    limit = int(math.sqrt(n)) + 1
    for i in range(3, limit, 2):
        if n % i == 0:
            return False
            
    return True

# --- Testing the function ---
numbers = [1, 2, 11, 15, 19, 25, 97]
prime_list = [num for num in numbers if is_prime(num)]

print(f"Primes found: {prime_list}")
# Output: Primes found: [2, 11, 19, 97]
