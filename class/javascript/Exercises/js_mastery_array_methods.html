function allSame(arr) {
    // If reverse searching finds the first element at the end, 
    // it doesn't guarantee all middle elements are the same.
    // Better array method constraint approach:
    // (Though normally we'd use .every())
    for (let i = 1; i < arr.length; i++) {

    }
    return true;
}

42. Custom Join Implementation

Write a function customJoin(arr, sep) that perfectly mimics array.join without using the native join or toString methods.

View Solution ▼ 
function customJoin(arr, sep = ',') {
    if (arr.length === 0) return "";
    let result = String(arr[0] ?? "");
    for (let i = 1; i < arr.length; i++) {
        result += sep + String(arr[i] ?? "");
    }
    return result;
}
console.log(customJoin([1, null, 3], '-')); // "1--3"
Hard

44. Removing Duplicates (indexOf)

Write a function unique(arr) that uses a loop and indexOf to return a new array with duplicates removed from [1, 2, 2, 3, 1]. Do not use Sets.

View Solution ▼ 
function unique(arr) {
    const result = [];
    for (let i = 0; i < arr.length; i++) {
        if (result.indexOf(arr[i]) === -1) {
            result.push(arr[i]);
        }
    }
    return result;
}
console.log(unique([1, 2, 2, 3, 1])); // [1, 2, 3]

48. Extracting and Joining Domains

Given URLs ['http://google.com', 'https://bing.com'], write code using string slice/indexOf within a loop to extract just domains and join them: "google.com, bing.com".

View Solution ▼ 
const urls = ['http://google.com', 'https://bing.com'];
const domains = [];

for (let url of urls) {
    const slashIdx = url.indexOf('//');
    if (slashIdx !== -1) {
        // String slice, not array slice, but similar concept
        domains.push(url.slice(slashIdx + 2)); 
    }
}
console.log(domains.join(', ')); // "google.com, bing.com"
